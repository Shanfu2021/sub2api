- [ ] **Step 2: Add the corrected binary workflow**

Create `.github/workflows/build-selfhosted-binary.yml`:

```yaml
name: Build Self-Hosted Binary

on:
  push:
    branches:
      - my_feature
  workflow_dispatch:

permissions:
  contents: read

env:
  FORCE_JAVASCRIPT_ACTIONS_TO_NODE24: "true"

jobs:
  build-linux-amd64:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v6

      - name: Setup pnpm
        uses: pnpm/action-setup@v6
        with:
          version: 9

      - name: Setup Node.js
        uses: actions/setup-node@v6
        with:
          node-version: "20"
          cache: pnpm
          cache-dependency-path: frontend/pnpm-lock.yaml

      - name: Install frontend dependencies
        working-directory: frontend
        run: pnpm install --frozen-lockfile

      - name: Build frontend assets
        working-directory: frontend
        env:
          NODE_OPTIONS: --max-old-space-size=1536
        run: pnpm run build

      - name: Setup Go
        uses: actions/setup-go@v6
        with:
          go-version-file: backend/go.mod
          check-latest: false
          cache: true
          cache-dependency-path: backend/go.sum

      - name: Verify Go version
        run: |
          expected="$(awk '/^go / {print $2; exit}' backend/go.mod)"
          go version | grep -q "go${expected}"

      - name: Build embedded server binary
        working-directory: backend
        env:
          CGO_ENABLED: "0"
        run: |
          VERSION="$(./scripts/resolve-version.sh)"
          COMMIT="${GITHUB_SHA}"
          DATE="$(date -u +%Y-%m-%dT%H:%M:%SZ)"
          mkdir -p ../dist
          go build -trimpath -tags embed \
            -ldflags="-s -w -X main.Version=${VERSION} -X main.Commit=${COMMIT} -X main.Date=${DATE} -X main.BuildType=release" \
            -o ../dist/sub2api ./cmd/server

      - name: Package artifact
        run: |
          VERSION="$(cd backend && ./scripts/resolve-version.sh)"
          mkdir -p package
          cp dist/sub2api package/
          cp deploy/config.example.yaml package/
          printf '%s\n' "$VERSION" > package/VERSION
          sha256sum package/sub2api > package/sub2api.sha256
          tar -C package -czf "sub2api-linux-amd64-${VERSION}.tar.gz" .

      - name: Upload build artifact
        uses: actions/upload-artifact@v4
        with:
          name: sub2api-linux-amd64
          path: |
            dist/sub2api
            package/sub2api.sha256
            package/VERSION
            sub2api-linux-amd64-*.tar.gz
          retention-days: 14
```
